# Жесты вне клавиатуры — в развёрнутом xterm (клики там безвредны).
#
# Ждём в журнале по порядку:
#   перетаскивание (479,600)        — нажатие в точке касания, а не за порогом;
#   перетаскивание закончено;
#   правый клик (479,600);
#   прокрутка вниз ×4                — 300 px при шаге 75 px (пальцы вверх — колесо вниз, с 09.11);
#   ничего после «второй палец снова» — средняя точка прыгнула, щелчков нет;
#   перетаскивание (479,600);
#   события потеряны, жест сброшен;
#   тап (856,46) ×2                   — кнопка «⌨» в панели: после каждого
#                                       тапа «svkbd: есть/нет» меняется;
#   ни одного тапа не теряется и после «события потеряны»;
# и button[1]=down перед остановкой демона, button[1]=up после.

x echo '# перетаскивание вправо на 200 px'
d 0 6 455 320
s
w 50
g 0 664 320 10 30
u 0
s
w 500

x echo '# удержание 1 с'
d 0 7 455 320
s
w 1000
u 0
s
w 500

x echo '# прокрутка двумя пальцами вверх на 300 px'
d 0 8 379 373
d 1 9 569 373
s
w 50
g2 0 -160 10 30
u 1
s
w 200
x echo '# второй палец снова, далеко внизу'
d 1 10 569 700
s
w 200
u 0
u 1
s
w 500

x echo '# потеря событий посреди перетаскивания'
d 0 11 455 320
s
w 50
m 0 520 320
s
w 200
x DISPLAY=:0 xinput query-state 'Virtual core XTEST pointer' | grep 'button\[1\]'
x pkill -STOP -x kt-test
f 1500
u 0
s
w 200
x pkill -CONT -x kt-test
w 700
x DISPLAY=:0 xinput query-state 'Virtual core XTEST pointer' | grep 'button\[1\]'

x echo '# кнопка «⌨» в панели дважды: клавиатура пропадает и возвращается'
d 0 12 812 25
s
w 80
u 0
s
w 2000
x pidof svkbd-mobile-intl >/dev/null && echo "svkbd: есть" || echo "svkbd: нет"
d 0 13 812 25
s
w 80
u 0
s
w 2000
x pidof svkbd-mobile-intl >/dev/null && echo "svkbd: есть" || echo "svkbd: нет"
