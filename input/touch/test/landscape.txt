# Альбом: экран повёрнут (xrandr --rotate right) на 1920×1080, svkbd внизу на
# всю ширину (1920×385 в (0,695)). Запускать так: ROTATE=right ./run.sh landscape.txt
# До прогона: xterm развёрнут и в фокусе, svkbd на слое en.
#
# Сырые координаты логической точки (X,Y) при повороте right:
#   raw_y = X·1023/1919,  raw_x = (1079−Y)·1023/1079.
#
# Ждём в журнале демона по порядку (координаты — логические, в пикселях):
#   «тап (50,41)»            — открылось «Меню» jwm: снимок menu.png (меню у
#                              левого верхнего угла); следующий тап закрывает его;
#   «тап (999,400)»          — закрыли меню кликом в xterm;
#   «клавиатура: нажато (208,798)» и «клавиатура: отпущено» — клавиша q,
#                              на снимке typed.png в строке xterm стоит q;
#   «клавиатура: нажато (1828,961)» и «отпущено» — Backspace стёр q;
#   «прокрутка вниз» ×4      — два пальца вверх на 300 px (колесо вниз, с 09.11);
#   «прокрутка влево» ×3     — два пальца вправо на 225 px (колесо влево);
#   «правый клик (999,400)» — удержание 1 с;
#   «тап (1695,48)» ×2       — кнопка «⌨» в панели: после каждого тапа
#                              «svkbd: есть/нет» меняется (была показана —
#                              «нет», затем «есть»).
# Ни одного «правый клик» в клавиатуре, ни одного лишнего тапа.

x echo '# тап в левый верхний угол: Меню jwm'
d 0 1 985 27
s
w 80
u 0
s
w 800
x DISPLAY=:0 scrot -o menu.png && echo "снимок: menu.png"
x echo '# тап в xterm закрывает меню'
d 0 2 644 533
s
w 80
u 0
s
w 800

x echo '# тап по клавише q'
d 0 3 267 111
s
w 80
u 0
s
w 500
x DISPLAY=:0 scrot -o typed.png && echo "снимок: typed.png"
x echo '# тап по Backspace'
d 0 4 112 975
s
w 80
u 0
s
w 500

x echo '# прокрутка двумя пальцами вверх на 300 px'
d 0 5 549 427
d 1 6 549 533
s
w 50
g2 285 0 10 30
u 1
u 0
s
w 500

x echo '# прокрутка двумя пальцами вправо на 225 px'
d 0 7 644 320
d 1 8 644 427
s
w 50
g2 0 120 10 30
u 1
u 0
s
w 500

x echo '# удержание 1 с: правый клик'
d 0 9 644 533
s
w 1000
u 0
s
w 500

x echo '# кнопка «⌨» в панели дважды: клавиатура пропадает и возвращается'
d 0 10 978 904
s
w 80
u 0
s
w 2000
x pidof svkbd-mobile-intl >/dev/null && echo "svkbd: есть" || echo "svkbd: нет"
d 0 11 978 904
s
w 80
u 0
s
w 2000
x pidof svkbd-mobile-intl >/dev/null && echo "svkbd: есть" || echo "svkbd: нет"
